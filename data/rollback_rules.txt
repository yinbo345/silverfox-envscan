# SilverFoxGuard 勒索回滚引擎配置（可热更新）
#
# ===========================================================================
#  为什么单独出一个配置文件
# ===========================================================================
#  卡巴斯基的 System Watcher 官方文档明确写：受监控的事件类型、回滚策略、
#  处置动作（Action on detection / Action if can be rolled back）全部可配。
#  本文件对齐这一做法：调参只需改这个文本文件，不必重新编译/签名/安装。
#
#  加载时机：主防服务启动时读 data\rollback_rules.txt。
#           （候选路径：<安装目录>\data\ 、<安装目录>\ 、C:\ProgramData\SilverFoxGuard\）
#  文件缺失或单条非法 → 该项回退内置默认值，不阻断启动。
# ===========================================================================
#
#  格式：key=value，# 之后为注释，空行忽略
# ===========================================================================
#
#  ---------------------------------------------------------------------------
#  【快照缓存】
#  ---------------------------------------------------------------------------
#  总容量上限（MB）。超出后按 LRU 淘汰最旧的快照。
#
#  ⚠️ 2026-09-18 由 512 下调到 160（维护者实测反馈"快照占用太大"）。
#     配合下方「风险分层建快照」后，正常使用下的实际占用远低于此上限；
#     160MB 是"极端场景兜底"，而不是日常预期值。
max_cache_mb = 160

#  单文件快照体积上限（MB）。超过此值的文件不备份
#  ——避免一个虚拟机镜像把整个缓存吃光。
#
#  ⚠️ 2026-09-18 由 32 下调到 4。依据：用户真实文档 99% 在 4MB 以内，
#     而 4MB 以上多为视频/镜像/虚拟机盘——那些即便被加密，用户也大概率
#     不会指望"原样回滚"。收紧后单文件占用上限降到原来的 1/8。
max_file_mb = 4

#  ---------------------------------------------------------------------------
#  【风险分层建快照】（2026-09-18 新增，快照瘦身的核心）
#  ---------------------------------------------------------------------------
#  为什么需要：旧策略是"任何改写都建快照"，于是普通软件每次保存文档都存一份
#  （Word 自动保存、IDE 频繁落盘、浏览器缓存写入……），占用迅速堆积，还会把
#  更早的、真正有价值的快照挤掉。
#
#  开启后只在下列任一条件成立时才建快照：
#    ① 文件被改成**陌生后缀**（勒索加密后必然改名，正常软件保存不改扩展名）；
#    ② 该文件所在目录在 burst_window_sec 秒内已被改写达 burst_file_trigger 次
#       （这片区域正在被批量动，风险升高）；
#    ③ 文件属于**高价值文档**（office/pdf/图片/音视频/代码），且是本轮首次改写。
#
#  关闭（设 0）则回到旧行为：任何改写都建快照，占用量大。
risk_tiered_snapshot = 1

#  目录级"热度"观察窗口（秒）与触发次数。
#  依据：正常软件极少在 30 秒内对同一目录做 5 次以上的**不同文件**改写；
#        一旦达到，说明发生了批量操作（无论善恶意），此后该目录的改写全部建快照。
burst_window_sec = 30
burst_file_trigger = 5

#  是否只对高价值文档做"首次改写即存"（判据③）。
#  设 0 则连普通类型的首次改写也不存，占用更省但保护面更窄。
high_value_only = 1

#  ---------------------------------------------------------------------------
#  【勒索判定：滑动窗口】
#  ---------------------------------------------------------------------------
#  为什么要"窗口 + 多信号"而不是简单计数：
#  正常软件（编译器、解压工具、批量重命名）也会在短时间内改大量文件，
#  只看数量必然误报。业界共识是必须联合"高熵随机后缀改名"与"写勒索说明"
#  等旁证才能定性。窗口越短越灵敏但越容易误判，10 秒是折中值。
window_seconds = 10

#  窗口内被改写的文件数达到该值 → 触发"批量改写"信号
files_threshold = 25

#  窗口内"改成高熵随机后缀"的次数达到该值 → 触发"改名"信号
rename_threshold = 10

#  窗口内出现勒索说明文件（README / HOW TO DECRYPT / 解密说明）的次数
note_threshold = 2

#  ---------------------------------------------------------------------------
#  【熵值阈值】
#  ---------------------------------------------------------------------------
#  Shannon 熵，单位 bit/byte，范围 0..8。
#  加密数据、压缩数据的典型熵值在 7.6 以上；普通文档约 3~6。
#  ⚠️ 单独看熵值**不足以**判定勒索（zip/7z/jpg/mp4 本身就接近 8.0），
#     本引擎只在配合"批量改动 + 陌生后缀改名"时才采用该信号。
entropy_threshold = 7.6

#  ---------------------------------------------------------------------------
#  【处置动作】（对齐卡巴的两档配置）
#  ---------------------------------------------------------------------------
#  Action on detection：检测到勒索后是否终止肇事进程。
#  1 = 终止（推荐，卡巴的 Action on detection 默认即"终止应用"）
#  0 = 不终止（仅回滚文件，适用于调试）
terminate_on_detect = 1

#  Action if can be rolled back：是否启用回滚。
#  本引擎的存在意义即回滚，故设为 0 时整个引擎不启动监控线程（省资源）。
enabled = 1

#  ---------------------------------------------------------------------------
#  【已知的局限性（诚实标注，不做虚假承诺）】
#  ---------------------------------------------------------------------------
#  1. 本引擎是**纯用户态**实现，不像卡巴那样有内核微过滤驱动在文件系统层拦写。
#     因此存在"被加密内容抢先写入"的竞态窗口（实测通知延迟约 1ms 量级）。
#     缓解手段是周期性基线巡逻：对尚无快照的受保护文件预先建立基线，
#     即使某次写前快照来晚了，也还能从更早的基线恢复。
#  2. 卡巴官方文档也承认回滚存在边界："有些情况下应用的活动无法回滚，
#     例如应用被 Intrusion Prevention 组件检测到时。"
#     本引擎对应的边界是：**没有快照的文件一律不可恢复**，会在报告中
#     如实列出清单，绝不尝试"猜测还原"（那是数据破坏，不是恢复）。
#  3. 快照缓存有容量上限，超出即淘汰最旧。淘汰掉的文件在后续勒索中
#     将不可恢复——这是"占盘"与"保护窗口"之间的必然取舍。
# ===========================================================================
